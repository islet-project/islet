6/// Keeps all platform token tag values based on RSS
7pub mod token_tag {
8    /* Claims */
9    pub const CCA_PLAT_CHALLENGE: u32 = 10;
10    pub const CCA_PLAT_INSTANCE_ID: u32 = 256;
11    pub const CCA_PLAT_PROFILE: u32 = 265;
12    pub const CCA_PLAT_SECURITY_LIFECYCLE: u32 = 2395;
13    pub const CCA_PLAT_IMPLEMENTATION_ID: u32 = 2396;
14    pub const CCA_PLAT_SW_COMPONENTS: u32 = 2399;
15    pub const CCA_PLAT_VERIFICATION_SERVICE: u32 = 2400;
16    pub const CCA_PLAT_CONFIGURATION: u32 = 2401;
17    pub const CCA_PLAT_HASH_ALGO_DESC: u32 = 2402;
18
19    /* Software components */
20    pub const CCA_SW_COMP_TITLE: u32 = 1;
21    pub const CCA_SW_COMP_MEASUREMENT_VALUE: u32 = 2;
22    pub const CCA_SW_COMP_VERSION: u32 = 4;
23    pub const CCA_SW_COMP_SIGNER_ID: u32 = 5;
24    pub const CCA_SW_COMP_HASH_ALGORITHM: u32 = 6;

27fn encode_measurement(measurement: &Measurement) -> Value {
28    let mut map: Vec<(Value, Value)> = Vec::with_capacity(5);
29
30    map.push((
31        Value::Integer(token_tag::CCA_SW_COMP_TITLE.into()),
32        Value::Text(measurement.metadata.sw_type.clone()),
33    ));
34    map.push((
35        Value::Integer(token_tag::CCA_SW_COMP_HASH_ALGORITHM.into()),
36        Value::Text(measurement.metadata.algorithm.into()),
37    ));
38    map.push((
39        Value::Integer(token_tag::CCA_SW_COMP_MEASUREMENT_VALUE.into()),
40        Value::Bytes(measurement.value.to_vec()),
41    ));
42    map.push((
43        Value::Integer(token_tag::CCA_SW_COMP_VERSION.into()),
44        Value::Text(measurement.metadata.sw_version.clone()),
45    ));
46    map.push((
47        Value::Integer(token_tag::CCA_SW_COMP_SIGNER_ID.into()),
48        Value::Bytes(measurement.metadata.signer_id.to_vec()),
49    ));
50
51    Value::Map(map)
52}
53
54pub fn encode_measurements(measurements: &[Measurement]) -> Value {
55    let mut array: Vec<Value> = Vec::with_capacity(measurements.len());
56
57    for measurement in measurements {
58        array.push(encode_measurement(measurement));
59    }
60
61    Value::Array(array)
62}
