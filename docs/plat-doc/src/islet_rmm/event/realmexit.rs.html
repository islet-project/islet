6/// Handles 'RET_TO_RMM' cases from trap::handle_lower_exception()
7///
8/// It can't cover all of RmiRecExitReason types.
9/// Because the following types of RmiRecExitReason
10/// are set to ExitReason using RSI:
11/// - RMI_EXIT_RIPAS_CHANGE
12/// - RMI_EXIT_HOST_CALL
13#[derive(Debug)]
14#[repr(usize)]
15pub enum RecExitReason {
16    Sync(ExitSyncType),
17    IRQ = 1,
18    FIQ = 2,
19    PSCI = 3,
20    SError = 4,
21    Undefined = REC_EXIT_REASON_MASK, // fixed, 0b1111
22}
23
24impl Into<u64> for RecExitReason {
25    fn into(self) -> u64 {
26        match self {
27            RecExitReason::Sync(exit_sync_type) => exit_sync_type.into(),
28            RecExitReason::IRQ => 1,
29            RecExitReason::FIQ => 2,
30            RecExitReason::PSCI => 3,
31            RecExitReason::SError => 4,
32            RecExitReason::Undefined => 7,
33        }
34    }
35}
36
37impl From<usize> for RecExitReason {
38    fn from(num: usize) -> Self {
39        match num & REC_EXIT_REASON_MASK {
40            0 => RecExitReason::Sync(ExitSyncType::from(num)),
41            1 => RecExitReason::IRQ,
42            2 => RecExitReason::FIQ,
43            3 => RecExitReason::PSCI,
44            4 => RecExitReason::SError,
45            _ => RecExitReason::Undefined,
46        }
47    }
48}

50/// Fault Status Code only for the 'RecExitReason::Sync'
51/// it's a different from trap::Syndrome
52#[derive(Debug, PartialEq)]
53#[repr(usize)]
54pub enum ExitSyncType {
55    RSI = 1 << EXIT_SYNC_TYPE_SHIFT,
56    DataAbort = 2 << EXIT_SYNC_TYPE_SHIFT,
57    InstAbort = 3 << EXIT_SYNC_TYPE_SHIFT,
58    WFx = 4 << EXIT_SYNC_TYPE_SHIFT,
59    Undefined = EXIT_SYNC_TYPE_MASK, // fixed, 0b1111_0000
60}
61
62impl Into<u64> for ExitSyncType {
63    fn into(self) -> u64 {
64        self as u64
65    }
66}
67
68impl From<usize> for ExitSyncType {
69    fn from(num: usize) -> Self {
70        let masked_val = (num & EXIT_SYNC_TYPE_MASK) >> EXIT_SYNC_TYPE_SHIFT;
71        match masked_val {
72            1 => ExitSyncType::RSI,
73            2 => ExitSyncType::DataAbort,
74            3 => ExitSyncType::InstAbort,
75            4 => ExitSyncType::WFx,
76            _ => ExitSyncType::Undefined,
77        }
78    }
