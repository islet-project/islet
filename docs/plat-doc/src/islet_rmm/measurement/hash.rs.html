10pub struct HashWrapper {
11    pub hash_func: Box<dyn DynDigest>,
12}
13
14impl HashWrapper {
15    pub fn hash(&mut self, data: impl AsRef<[u8]>) {
16        self.hash_func.update(data.as_ref());
17    }
18
19    pub fn hash_u8(&mut self, data: u8) {
20        self.hash_func.update(data.to_le_bytes().as_slice());
21    }
22
23    pub fn hash_u16(&mut self, data: u16) {
24        self.hash_func.update(data.to_le_bytes().as_slice());
25    }
26
27    pub fn hash_u32(&mut self, data: u32) {
28        self.hash_func.update(data.to_le_bytes().as_slice());
29    }
30
31    pub fn hash_u64(&mut self, data: u64) {
32        self.hash_func.update(data.to_le_bytes().as_slice());
33    }
34
35    pub fn hash_usize(&mut self, data: usize) {
36        self.hash_func.update(data.to_le_bytes().as_slice());
37    }
38
39    pub fn hash_u64_array(&mut self, array: &[u64]) {
40        for el in array.iter() {
41            self.hash_func.update(el.to_le_bytes().as_slice());
42        }
43    }
44
45    fn finish(&mut self, mut out: impl AsMut<[u8]>) -> Result<(), MeasurementError> {
46        self.hash_func
47            .finalize_into_reset(&mut out.as_mut()[0..self.hash_func.output_size()])
48            .map_err(|_| MeasurementError::OutputBufferTooSmall)
49    }
50}

52pub struct Hasher {
53    factory: Box<dyn Fn() -> Box<dyn DynDigest>>,
54    block_size: usize,
55}
56
57impl Hasher {
58    pub fn from_hash_algo(hash_algo: u8) -> Result<Self, MeasurementError> {
59        let factory: Box<dyn Fn() -> Box<dyn DynDigest>> = match hash_algo {
60            HASH_ALGO_SHA256 => Box::new(|| Box::new(Sha256::new())),
61            HASH_ALGO_SHA512 => Box::new(|| Box::new(Sha512::new())),
62            _ => return Err(MeasurementError::InvalidHashAlgorithmValue(hash_algo)),
63        };
64
65        let block_size = match hash_algo {
66            HASH_ALGO_SHA256 => <Sha256 as Digest>::output_size(),
67            HASH_ALGO_SHA512 => <Sha512 as Digest>::output_size(),
68            _ => return Err(MeasurementError::InvalidHashAlgorithmValue(hash_algo)),
69        };
70
71        Ok(Self {
72            factory,
73            block_size,
74        })
75    }
76
77    pub fn hash_fields_into(
78        &self,
79        out: impl AsMut<[u8]>,
80        f: impl Fn(&mut HashWrapper),
81    ) -> Result<(), MeasurementError> {
82        let mut wrapper = HashWrapper {
83            hash_func: (self.factory)(),
84        };
85        f(&mut wrapper);
86        wrapper.finish(out)
87    }
88
89    pub fn hash_object_into(
90        &self,
91        obj: &dyn Hashable,
92        mut out: impl AsMut<[u8]>,
93    ) -> Result<(), MeasurementError> {
94        obj.hash(self, out.as_mut())
95    }
96
97    pub fn output_size(&self) -> usize {
98        self.block_size
99    }
100}
