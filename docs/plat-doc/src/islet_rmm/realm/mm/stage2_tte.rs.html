41pub fn mapping_size(level: usize) -> usize {
42    match level {
43        3 => 1 << S2TTE::ADDR_BLK_L3.trailing_zeros(), // 4096
44        2 => 1 << S2TTE::ADDR_BLK_L2.trailing_zeros(),
45        1 => 1 << S2TTE::ADDR_BLK_L1.trailing_zeros(),
46        0 => 1 << S2TTE::ADDR_BLK_L0.trailing_zeros(),
47        _ => unreachable!(),
48    }
49}

51pub fn level_mask(level: usize) -> Option<u64> {
52    match level {
53        3 => Some(S2TTE::ADDR_BLK_L3),
54        2 => Some(S2TTE::ADDR_BLK_L2),
55        1 => Some(S2TTE::ADDR_BLK_L1),
56        0 => Some(S2TTE::ADDR_BLK_L0),
57        _ => None,
58    }
59}

61fn is_block_level(level: usize) -> bool {
62    level >= RTT_MIN_BLOCK_LEVEL && level < RTT_PAGE_LEVEL
63}

65define_bits!(
66    S2TTE,
67    HIPAS[58 - 58], // Host IPA State (HIPAS)
68    RIPAS[57 - 56], // Realm IPA State (RIPAS)
69    NS[55 - 55], // DDI0615A: For a Block or Page descriptor fetched for stage 2 in the Realm Security state, bit 55 is the NS field. if set, it means output address is in NS PAS.
70    XN[54 - 54],
71    CONT[52 - 52],
72    // https://armv8-ref.codingbelief.com/en/chapter_d4/d43_1_vmsav8-64_translation_table_descriptor_formats.html
73    ADDR_BLK_L0[47 - 39],      // block descriptor; level 0 w/o concatenation
74    ADDR_BLK_L1[47 - 30],      // block descriptor; level 1
75    ADDR_BLK_L2[47 - 21],      // block descriptor; level 2
76    ADDR_BLK_L3[47 - 12],      // page descriptor; level 3
77    ADDR_TBL_OR_PAGE[47 - 12], // table descriptor(level 0-2) || page descriptor(level3)
78    AF[10 - 10],
79    SH[9 - 8],      // pte_shareable
80    S2AP[7 - 6],    // pte_access_perm
81    MEMATTR[5 - 2], // pte_mem_attr
82    DESC_TYPE[1 - 0],
83    TYPE[1 - 1], // pte_type ; block(0) or table(1)
84    VALID[0 - 0]
85);

87impl S2TTE {
88    pub fn get_s2tte(
89        rd: &Rd,
90        ipa: usize,
91        level: usize,
92        error_code: Error,
93    ) -> Result<(S2TTE, usize), Error> {
94        let (mut s2tte, last_level) = rd
95            .s2_table()
96            .lock()
97            .ipa_to_pte(GuestPhysAddr::from(ipa), level)
98            .ok_or(error_code)?;
99
100        if ipa < rd.ipa_size() && !rd.addr_in_par(ipa) && s2tte == 0 {
101            s2tte |= S2TTE::NS;
102        }
103
104        Ok((S2TTE::from(s2tte as usize), last_level))
105    }

107    pub fn is_host_ns_valid(&self, level: usize) -> bool {
108        let tmp = S2TTE::new(!0);
109        let addr_mask = match level {
110            1 => tmp.get_masked(S2TTE::ADDR_BLK_L1),
111            2 => tmp.get_masked(S2TTE::ADDR_BLK_L2),
112            3 => tmp.get_masked(S2TTE::ADDR_BLK_L3),
113            _ => return false,
114        };
115        let mask = addr_mask
116            | tmp.get_masked(S2TTE::MEMATTR)
117            | tmp.get_masked(S2TTE::S2AP)
118            | tmp.get_masked(S2TTE::SH);
119
120        if (self.get() & !mask) != 0 {
121            return false;
122        }
123
124        if self.get_masked_value(S2TTE::MEMATTR) == memattr::FWB_RESERVED {
125            return false;
126        }
127
128        if self.get_masked_value(S2TTE::SH) == shareable::RESERVED {
129            return false;
130        }
131
132        true
133    }

135    pub fn is_unassigned(&self) -> bool {
136        self.get_masked_value(S2TTE::DESC_TYPE) == desc_type::LX_INVALID
137            && self.get_masked_value(S2TTE::HIPAS) == hipas::UNASSIGNED
138            && self.get_masked_value(S2TTE::NS) == 0
139    }
140
141    pub fn is_unassigned_empty(&self) -> bool {
142        self.is_unassigned() && self.get_masked_value(S2TTE::RIPAS) == ripas::EMPTY
143    }

145    pub fn is_unassigned_destroyed(&self) -> bool {
146        self.is_unassigned() && self.get_masked_value(S2TTE::RIPAS) == ripas::DESTROYED
147    }

149    pub fn is_unassigned_ram(&self) -> bool {
150        self.is_unassigned() && self.get_masked_value(S2TTE::RIPAS) == ripas::RAM
151    }
152
153    pub fn is_unassigned_ns(&self) -> bool {
154        self.get_masked_value(S2TTE::DESC_TYPE) == desc_type::LX_INVALID
155            && self.get_masked_value(S2TTE::HIPAS) == hipas::UNASSIGNED
156            && self.get_masked_value(S2TTE::NS) != 0
157    }

159    pub fn is_destroyed(&self) -> bool {
160        self.get_masked_value(S2TTE::DESC_TYPE) == desc_type::LX_INVALID
161            && self.get_masked_value(S2TTE::RIPAS) == ripas::DESTROYED
162            && self.get_masked_value(S2TTE::NS) == 0
163    }

165    pub fn is_assigned(&self) -> bool {
166        self.get_masked_value(S2TTE::NS) == 0
167            && self.get_masked_value(S2TTE::HIPAS) == hipas::ASSIGNED
168    }

170    pub fn is_assigned_invalid(&self) -> bool {
171        self.get_masked_value(S2TTE::NS) == 0
172            && self.get_masked_value(S2TTE::DESC_TYPE) == desc_type::LX_INVALID
173            && self.get_masked_value(S2TTE::HIPAS) == hipas::ASSIGNED
174    }
175
176    pub fn is_assigned_empty(&self) -> bool {
177        self.is_assigned_invalid() && self.get_masked_value(S2TTE::RIPAS) == ripas::EMPTY
178    }
179
180    pub fn is_assigned_destroyed(&self) -> bool {
181        self.is_assigned_invalid() && self.get_masked_value(S2TTE::RIPAS) == ripas::DESTROYED
182    }

184    pub fn is_assigned_ram(&self, level: usize) -> bool {
185        if self.get_masked_value(S2TTE::NS) != 0
186            || self.get_masked_value(S2TTE::RIPAS) != ripas::RAM
187        {
188            return false;
189        }
190        let desc_type = self.get_masked_value(S2TTE::DESC_TYPE);
191        if (level == RTT_PAGE_LEVEL && desc_type == desc_type::L3_PAGE)
192            || (is_block_level(level) && desc_type == desc_type::L012_BLOCK)
193        {
194            return true;
195        }
196        false
197    }
198
199    pub fn is_assigned_ns(&self, level: usize) -> bool {
200        if self.get_masked_value(S2TTE::NS) == 0
201            || self.get_masked_value(S2TTE::HIPAS) != hipas::ASSIGNED
202        {
203            return false;
204        }
205        let desc_type = self.get_masked_value(S2TTE::DESC_TYPE);
206        if (level == RTT_PAGE_LEVEL && desc_type == desc_type::L3_PAGE)
207            || (is_block_level(level) && desc_type == desc_type::L012_BLOCK)
208        {
209            return true;
210        }
211        false
212    }

214    pub fn has_ripas(&self, level: usize) -> bool {
215        self.get_masked_value(S2TTE::NS) == 0 && !self.is_table(level)
216    }

218    // level should be the value returned in page table walking
219    // (== the last level that has been reached)
220    pub fn is_table(&self, level: usize) -> bool {
221        (level < RTT_PAGE_LEVEL)
222            && self.get_masked_value(S2TTE::DESC_TYPE) == desc_type::L012_TABLE
223            && self.get_masked_value(S2TTE::NS | S2TTE::HIPAS | S2TTE::RIPAS) == 0x0
224    }
225
226    pub fn is_ripas(&self) -> bool {
227        (self.get_masked_value(S2TTE::DESC_TYPE) != desc_type::LX_INVALID)
228            && (self.get_ripas() != ripas::RAM)
229    }

231    pub fn addr_as_block(&self, level: usize) -> PhysAddr {
232        match level {
233            1 => PhysAddr::from(self.get_masked(S2TTE::ADDR_BLK_L1)),
234            2 => PhysAddr::from(self.get_masked(S2TTE::ADDR_BLK_L2)),
235            3 => PhysAddr::from(self.get_masked(S2TTE::ADDR_BLK_L3)),
236            _ => unreachable!(),
237        }

244    pub fn is_live(&self, _level: usize) -> bool {
245        // live tte: ASSIGNED, ASSIGNED_NS, TABLE
246        self.get_masked_value(S2TTE::DESC_TYPE) != desc_type::LX_INVALID
247            || self.get_masked_value(S2TTE::HIPAS) == hipas::ASSIGNED

251    pub fn is_homogeneous(entries: &mut Iter<'_, entry::Entry>, level: usize) -> bool {
252        let mut hipas = 0;
253        let mut ripas = 0;
254        let mut desc_type = 0;
255        let mut pa = 0;
256        let mut attr: u64 = 0;
257        let mut ns = 0;
258        let mut first = true;
259        let map_size = mapping_size(level) as u64;
260        for entry in entries {
261            let s2tte = S2TTE::new(entry.pte());
262            if first {
263                desc_type = s2tte.get_masked_value(S2TTE::DESC_TYPE);
264                hipas = s2tte.get_masked_value(S2TTE::HIPAS);
265                ripas = s2tte.get_masked_value(S2TTE::RIPAS);
266                ns = s2tte.get_masked_value(S2TTE::NS);
267                first = false;
268                if s2tte.is_assigned_ns(level) | s2tte.is_assigned() {
269                    // level is 2 or 3
270                    if level != 2 && level != 3 {
271                        return false;
272                    }
273                    pa = s2tte.addr_as_block(level).into(); //XXX: check this again
274                    attr = entry.pte() & !level_mask(level).unwrap_or(0);
275                    // output of first entry is algned to parent's mapping size
276                    if (pa & !level_mask(level - 1).unwrap_or(0)) != 0 {
277                        return false;
278                    }
279                }
280                continue;
281            }
282
283            if desc_type != s2tte.get_masked_value(S2TTE::DESC_TYPE) {
284                return false;
285            } else if s2tte.is_assigned_ns(level) {
286                // addr is contiguous
287                pa += map_size;
288                if pa != s2tte.addr_as_block(level).into() {
289                    return false;
290                }
291                // attributes are identical
292                if attr != s2tte.get() & !level_mask(level).unwrap_or(0) {
293                    return false;
294                }
295            } else if s2tte.is_assigned() {
296                // addr is contiguous
297                pa += map_size;
298                if pa != s2tte.addr_as_block(level).into() {
299                    return false;
300                }
301                // ripas is identical
302                if ripas != s2tte.get_masked_value(S2TTE::RIPAS) {
303                    return false;
304                }
305            } else if s2tte.is_unassigned_ns() {
306                // ns is always 1
307                if ns != s2tte.get_masked_value(S2TTE::NS) {
308                    return false;
309                }
310                // hipas is always UNASSIGNED
311                if hipas != s2tte.get_masked_value(S2TTE::HIPAS) {
312                    return false;
313                }
314            } else if s2tte.is_unassigned() {
315                // hipas is always UNASSIGNED
316                if hipas != s2tte.get_masked_value(S2TTE::HIPAS) {
317                    return false;
318                }
319                // ripas is identical
320                if ripas != s2tte.get_masked_value(S2TTE::RIPAS) {
321                    return false;
322                }
323            }
324        }
325        true
326    }
