5/// Supported public dak hash algorithms.
6#[allow(dead_code)]
7#[derive(Debug)]
8enum HashAlgo {
9    Sha256,
10    Sha384,
11    Sha512,
12}
13
14fn calculate_hash(data: Vec<u8>, algo: HashAlgo) -> Vec<u8> {
15    match algo {
16        HashAlgo::Sha256 => {
17            let mut hasher = Sha256::new();
18            hasher.update(data);
19            hasher.finalize().to_vec()
20        }
21        HashAlgo::Sha384 => {
22            let mut hasher = Sha384::new();
23            hasher.update(data);
24            hasher.finalize().to_vec()
25        }
26        HashAlgo::Sha512 => {
27            let mut hasher = Sha512::new();
28            hasher.update(data);
29            hasher.finalize().to_vec()
30        }
31    }
32}

34pub(super) fn get_realm_public_key_hash(key: Vec<u8>) -> Vec<u8> {
35    let priv_dak = p384::SecretKey::from_slice(&key).unwrap();
36    let public_dak = priv_dak.public_key().to_sec1_bytes().to_vec();
37    let public_dak_cose = cose::ec_public_key_sec1_to_cose(&public_dak);
38
39    calculate_hash(public_dak_cose, HashAlgo::Sha256)
