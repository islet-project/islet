92pub fn load() -> core::result::Result<(), Error> {
93    debug!("Configuring RMM with EL3 manifest");
94    let guard: SpinlockGuard<'_, _> = RMM_SHARED_BUFFER_LOCK.lock();
95    let manifest = assume_safe::<RmmManifest>(*guard)?;
96    let struct_size = core::mem::size_of::<DramBank>();
97    let mut dram_vec = config::NS_DRAM_REGIONS.lock();
98    debug!("version: {:?}", manifest.version);
99    debug!("num_banks: {:x}", manifest.num_banks);
100    if manifest.version != EL3_IFC_VERSION {
101        panic!(
102            "manifest version {:X} not supported. requires {:X}",
103            manifest.version, EL3_IFC_VERSION
104        );
105    }
106    let mut bank_ptr = manifest.banks_ptr as usize;
107    let mut max_base = 0;
108    for i in 0..manifest.num_banks {
109        let bank = assume_safe::<DramBank>(bank_ptr)?;
110        debug!(
111            "NS_DRAM[{:?}]: {:X}-{:X} (size:{:X})",
112            i,
113            bank.base,
114            bank.base + bank.size,
115            bank.size
116        );
117        dram_vec.push(core::ops::Range {
118            start: bank.base as usize,
119            end: (bank.base + bank.size) as usize,
120        });
121        bank_ptr += struct_size;
122        if bank.base < max_base {
123            panic!("Islet only accepts an ordered bank list");
124        }
125        max_base = bank.base;
126    }
127    Ok(())
128}
