26pub fn setup_el3_ifc(el3_shared_buf: u64) {
27    trace!("Setup EL3 interface");
28
29    // In case of warm boot, el3_shared_buf is delivered with 0x0
30    if el3_shared_buf == 0x0 {
31        warn!("el3_sshared_buf address is 0x0. Won't initialize again.");
32        return;
33    }
34
35    {
36        // limit the scope of lock to this scope
37        // to avoid spinning forever in the subsequent functions
38        let mut guard: SpinlockGuard<'_, _> = RMM_SHARED_BUFFER_LOCK.lock();
39        *guard = el3_shared_buf as usize;
40    }
41    let _ = manifest::load();
42    iface::get_realm_attest_key();
43    iface::get_plat_token();
44    iface::get_vhuks();
45}
